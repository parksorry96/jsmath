```typescript
@Injectable()
export class RedisStreamService implements OnModuleInit, OnModuleDestroy {
  private running = true;

  async onModuleInit() {
    const streams = [
      'stream:ocr:completed', 'stream:ocr:failed',
      'stream:analysis:completed', 'stream:analysis:failed',
      'stream:photo:analysis:completed', 'stream:photo:analysis:failed',
      'stream:photo:rubric:completed', 'stream:photo:rubric:failed',
    ];

    for (const stream of streams) {
      await this.redis.xgroup('CREATE', stream, 'lms-api', '0', 'MKSTREAM')
        .catch(() => {}); // group already exists
    }

    this.consumeLoop(streams);
  }

  private async consumeLoop(streams: string[]) {
    while (this.running) {
      const results = await this.redis.xreadgroup(
        'GROUP', 'lms-api', `consumer-${process.pid}`,
        'COUNT', '10', 'BLOCK', '5000',
        'STREAMS', ...streams, ...streams.map(() => '>'),
      );

      for (const [stream, messages] of results ?? []) {
        for (const [id, fields] of messages) {
          await this.dispatch(stream, JSON.parse(fields.data));
          await this.redis.xack(stream, 'lms-api', id);
        }
      }
    }
  }
}
```

```typescript
async saveRevisionAndUpdate(
  problemId: string,
  updateData: Partial<UpdateProblemDto>,
  changedById: string,
  changedByRole: string,
): Promise<void> {
  const current = await this.prisma.problem.findUniqueOrThrow({
    where: { id: problemId },
  });

  const changedFields = Object.keys(updateData).filter(
    k => updateData[k] !== current[k]
  );
  if (changedFields.length === 0) return;

  const latestRev = await this.prisma.problemRevision.findFirst({
    where: { problemId },
    orderBy: { revisionNumber: 'desc' },
    select: { revisionNumber: true },
  });

  await this.prisma.$transaction([
    this.prisma.problemRevision.create({
      data: {
        problemId,
        revisionNumber: (latestRev?.revisionNumber ?? 0) + 1,
        snapshot: current as unknown as Prisma.JsonObject,
        changeType: this.classifyChangeType(changedFields),
        changedFields,
        changedById,
        changedByRole,
        changeSource: 'manual',
      },
    }),
    this.prisma.problem.update({
      where: { id: problemId },
      data: updateData,
    }),
  ]);
}
```

Core computation in `problem-statistics.service.ts`:

```typescript
async recomputeForProblem(problemId: string): Promise<void> {
  const answers = await this.prisma.submissionAnswer.findMany({
    where: { problemId, isCorrect: { not: null } },
    select: {
      isCorrect: true, score: true, studentAnswer: true,
      submission: { select: { score: true, maxScore: true } },
    },
  });
  if (answers.length === 0) return;

  const n = answers.length;
  const correctCount = answers.filter(a => a.isCorrect).length;
  const pValue = correctCount / n;

  // Discrimination: top/bottom 27% by total score
  const sorted = answers
    .filter(a => a.submission.maxScore && a.submission.maxScore > 0)
    .sort((a, b) => {
      const rA = (a.submission.score ?? 0) / a.submission.maxScore!;
      const rB = (b.submission.score ?? 0) / b.submission.maxScore!;
      return rB - rA;
    });
  const cutoff = Math.floor(sorted.length * 0.27);
  const discIndex = cutoff > 0
    ? (sorted.slice(0, cutoff).filter(a => a.isCorrect).length / cutoff) -
      (sorted.slice(-cutoff).filter(a => a.isCorrect).length / cutoff)
    : null;

  await this.prisma.problemStatistics.upsert({
    where: { problemId },
    create: { problemId, attemptCount: n, correctCount, pValue,
              discriminationIndex: discIndex, isSufficientN: n >= 30,
              lastComputedAt: new Date() },
    update: { attemptCount: n, correctCount, pValue,
              discriminationIndex: discIndex, isSufficientN: n >= 30,
              lastComputedAt: new Date() },
  });
}
```

```typescript
const THRESHOLDS = {
  minAttempts: 30,
  pValueMin: 0.10,       // < 10% correct → likely content error
  pValueMax: 0.95,       // > 95% correct → no discrimination
  discriminationMin: 0.20,
} as const;

async evaluateAndFlag(problemId: string): Promise<void> {
  const stats = await this.prisma.problemStatistics.findUnique({ where: { problemId } });
  if (!stats || !stats.isSufficientN) return;

  const flags: string[] = [];
  if (stats.pValue !== null && stats.pValue < THRESHOLDS.pValueMin) flags.push('p_value_too_low');
  if (stats.pValue !== null && stats.pValue > THRESHOLDS.pValueMax) flags.push('p_value_too_high');
  if (stats.discriminationIndex !== null && stats.discriminationIndex < THRESHOLDS.discriminationMin)
    flags.push('low_discrimination');

  if (flags.length === 0) {
    // Promote to calibrated
    await this.prisma.problem.update({
      where: { id: problemId },
      data: { reviewStatus: 'calibrated' },
    });
  } else {
    await this.prisma.problem.update({
      where: { id: problemId },
      data: { reviewStatus: 'flagged', flagReason: flags.join(',') },
    });
  }
}
```

```
calibrated ──(stats degrade)──→ flagged
flagged ──(teacher review)──→ approved | rejected | retired
```

## Task 9: Hybrid Search (RRF)

**Priority:** P2 — Search quality improvement
**Size:** S (1.5 days)
**Depends on:** Task 5 (tsvector must be populated)

**Files:**
- Modify: `apps/lms-api/src/problems/problems.service.ts`

**Context:** `findAll` uses `ILIKE` for keyword search (`problems.service.ts:211-213`). `semanticSearch` uses pgvector cosine distance separately (`problems.service.ts:295-351`). No combined ranking exists.

**Step 1: Add hybridSearch method**

```typescript
async hybridSearch(query: string, filters: ProblemSearchParams, take: number, skip: number) {
  const embedding = await this.embeddingService.embed(query);

  const results = await this.prisma.$queryRaw`
    WITH keyword_results AS (
      SELECT id,
             ts_rank(stem_tsv, plainto_tsquery('simple', ${query})) AS kw_score,
             ROW_NUMBER() OVER (ORDER BY ts_rank(stem_tsv, plainto_tsquery('simple', ${query})) DESC) AS kw_rank
      FROM ocr.problems
      WHERE stem_tsv @@ plainto_tsquery('simple', ${query})
        AND review_status IN ('approved', 'auto_approved', 'calibrated')
      LIMIT 100
    ),
    semantic_results AS (
      SELECT id,
             1 - (embedding <=> ${embedding}::vector) AS sem_score,
             ROW_NUMBER() OVER (ORDER BY embedding <=> ${embedding}::vector) AS sem_rank
      FROM ocr.problems
      WHERE embedding IS NOT NULL
        AND review_status IN ('approved', 'auto_approved', 'calibrated')
      LIMIT 100
    )
    SELECT COALESCE(k.id, s.id) AS id,
           COALESCE(1.0 / (60 + k.kw_rank), 0) +
           COALESCE(1.0 / (60 + s.sem_rank), 0) AS rrf_score
    FROM keyword_results k
    FULL OUTER JOIN semantic_results s ON k.id = s.id
    ORDER BY rrf_score DESC
    LIMIT ${take} OFFSET ${skip}
  `;

  // Fetch full problem data for returned IDs
  const ids = results.map(r => r.id);
  return this.prisma.problem.findMany({
    where: { id: { in: ids } },
    // preserve RRF order
  });
}
```

**Step 2: Route `findAll` queries through hybrid search**

When `q` parameter is provided and `searchMode` is `hybrid` (or default), delegate to `hybridSearch`. Keep `ILIKE` as `keyword`-only fallback.

**Verify:** Search for "이차방정식 판별식" — results should include problems with related concepts (semantic) even if exact terms don't match (keyword).

## Task 12: Blueprint Auto-Composition Algorithm

**Priority:** P2 — Core value of blueprint feature
**Size:** L (4 days)
**Depends on:** Task 11, Task 4 (usage log for exclusion)

**Files:**
- Create: `apps/lms-api/src/exam-blueprints/blueprint-composer.service.ts`

**Context:** Problem pool has rich metadata. Algorithm must select N problems matching blueprint constraints while maximizing diversity and coverage.

**Step 1: Algorithm — Greedy selection + Swap refinement**

```
1. Filter candidate pool (approved/calibrated, matching gradeLevel, not excluded)
2. Create slots from blueprint distributions (unit × difficulty × type)
3. Sort slots by constraint tightness (fewer candidates = higher priority)
4. Greedy fill: for each slot, pick best-fit candidate by composite score
5. Swap refinement: iterate up to 100 times, swap selected↔pool if it improves overall fit
6. Compute match score (0-1) and record any relaxations
```

Fit score components:
- Unit match: 40% (exact node > unitMajor > subject)
- Difficulty match: 25%
- Type match: 20%
- Source diversity: 10% (penalize same sourceFile)
- Concept coverage: 5%

**Step 2: Relaxation strategy (when pool doesn't perfectly match)**

Priority order:
1. Expand difficulty range ±1
2. Expand unit scope (unitMinor → unitMajor → subject)
3. Relax type constraint
4. Adjust slot count ±1

Each relaxation is recorded in `BlueprintGeneration.relaxations`.

**Step 3: Preview endpoint**

```
POST /exam-blueprints/:id/preview
Response: { feasibility: 'full'|'partial'|'insufficient', availableCounts: {...} }
```

**Step 4: Compose endpoint**

```
POST /exam-blueprints/:id/compose
Response: { problemIds: [...], matchScore: 0.92, relaxations: [...] }
```

**Verify:** Create a blueprint for 22 problems (수능 format). Run compose. Verify unit/difficulty distribution matches within 10%. Run again — verify different selection (randomized tie-breaking). Create blueprint with impossible constraints — verify `feasibility: 'insufficient'`.

```typescript
@WebSocketGateway({ namespace: '/class-monitor', cors: { origin: '*' } })
export class ClassMonitorGateway implements OnGatewayConnection, OnGatewayDisconnect {
  @WebSocketServer() server: Server;

  async handleConnection(client: Socket) {
    const token = client.handshake.auth?.token;
    const user = await this.jwtService.verifyAsync(token);
    if (user.role !== 'admin' && user.role !== 'teacher') {
      client.disconnect();
      return;
    }
    client.data.user = user;
  }

  @SubscribeMessage('join:assignment')
  handleJoin(client: Socket, { assignmentId }: { assignmentId: string }) {
    client.join(`assignment:${assignmentId}`);
    this.sendInitialStatus(client, assignmentId);
  }
}
```
